# request frame, scheduler event, session step, and sampled token.

^serve\.scheduler\.Scheduler\(.*\)\.(handle|acceptToken|tick|runComposedStep)$
^serve\.residency\.Residency\(.*\)\.(advance|restoreReady|restorePrefix|evict\w*)$
^runtime\.session\.Sessions\(.*\)\.(advance|clone|create|restorePrefix)$
^runtime\.(llama|qwen)_runner\.(LlamaRunner|QwenRunner)\.(stepBatch|evictSlot|restoreSlot)$
^runtime\.(llama|qwen)_(fused|hybrid)\.\w+\.(stepBatch|prefillBatch|evictSlot|restoreSlot)$
^runtime\.sampler\.(sample|samplePool)$
^serve\.openai\.\w+\.\w+$
